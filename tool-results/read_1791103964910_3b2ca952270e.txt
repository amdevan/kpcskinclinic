    11→// Slugify a title for use in /services/[slug] links
    12→export function slugify(s: string): string {
    13→  return s
    14→    .toLowerCase()
    15→    .replace(/[()]/g, "")
    16→    .replace(/[^a-z0-9]+/g, "-")
    17→    .replace(/^-+|-+$/g, "");
    18→}
